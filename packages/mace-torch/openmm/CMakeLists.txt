cmake_minimum_required(VERSION 3.20)
project(mace_openmm_native VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(mace CONFIG REQUIRED)
find_path(OPENMM_INCLUDE_DIR openmm/internal/CustomCPPForceImpl.h REQUIRED)
find_library(OPENMM_LIBRARY NAMES OpenMM REQUIRED)
add_library(mace_openmm SHARED MaceForce.cpp)
target_compile_features(mace_openmm PUBLIC cxx_std_17)
target_include_directories(mace_openmm PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>"
  "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/mace>" "${OPENMM_INCLUDE_DIR}")
target_link_libraries(mace_openmm PUBLIC mace::mace "${OPENMM_LIBRARY}")
set_target_properties(mace_openmm PROPERTIES EXPORT_NAME openmm INSTALL_RPATH_USE_LINK_PATH TRUE)
add_library(mace::openmm ALIAS mace_openmm)
install(TARGETS mace_openmm EXPORT mace-openmm-targets LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}")
install(FILES MaceForce.h DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/mace")
configure_package_config_file(mace-openmm-config.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/mace-openmm-config.cmake"
  INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/mace-openmm")
install(EXPORT mace-openmm-targets NAMESPACE mace:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/mace-openmm")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/mace-openmm-config.cmake" DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/mace-openmm")
set(MACE_OPENMM_TEST_DRIVER "" CACHE FILEPATH "Optional native integration driver")
if(MACE_OPENMM_TEST_DRIVER)
  add_executable(mace_openmm_driver "${MACE_OPENMM_TEST_DRIVER}")
  target_link_libraries(mace_openmm_driver PRIVATE mace_openmm)
endif()
