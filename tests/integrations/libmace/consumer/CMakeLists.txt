cmake_minimum_required(VERSION 3.20)
project(libmace_installed_consumer LANGUAGES CXX)
option(MACE_WITH_FORTRAN "Check the installed Fortran module" ON)
if(MACE_WITH_FORTRAN)
  enable_language(Fortran)
endif()
find_package(mace CONFIG REQUIRED)
add_executable(mace_libmace_driver ../driver.cpp)
target_compile_features(mace_libmace_driver PRIVATE cxx_std_17)
target_link_libraries(mace_libmace_driver PRIVATE mace::mace)
if(MACE_WITH_FORTRAN)
  add_executable(mace_libmace_fortran ../driver.f90)
  target_link_libraries(mace_libmace_fortran PRIVATE mace::fortran)
endif()
